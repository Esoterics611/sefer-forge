## הַנֹּסַח הָעִקָּרִי

[הַשָּׁמַיִם, בְּאֹפֶן שֶׁיִּהְיוּ כָּל גְּבוּלוֹת הָאָרֶץ כָּל אַחַד עַל מְקוֹמוֹ יָבוֹא לְקַבֵּל אוֹר הַשֶּׁפַע הָרוּחָנִיּוֹת מִן הַשָּׁמַיִם לְנֶגְדּוֹ וְכַנַּ"ל, וְאַחַר שֶׁאַתָּה הִצַּבְתָּ בְּאֹפֶן פְּרָטִי כָּל הַגְּבוּלוֹת אֶרֶץ, מִמֵּילָא קַיִץ וְחוֹרֶף אַתָּה יְצַרְתָּם לָהֶם בְּשִׁינּוּי אֹפְנֵי הַמְּקוֹמוֹת לְפִי מַעֲמָדָם אֲשֶׁר הִצַּבְתָּ אוֹתָם בָּאָרֶץ, וְכַנַּ"ל.]{dir="rtl"}

[**וְהִנֵּה**, זֶה הַמּוֹפֵת וְהָרְאָיָה מִקַּיִץ וְחוֹרֶף אֵינוֹ אֶלָּא לְאֻמּוֹת הָעוֹלָם, שֶׁנִּקְרָאִים עַם נָבָל, וְהֵם אֲשֶׁר נִאֲצוּ שִׁמְךָ לֵאמֹר עָזַב ה' אֶת הָאָרֶץ תַּחַת מִקְרֵי הַטֶּבַע, לִהְיוֹת כִּי שֹׁרֶשׁ נַפְשׁוֹת אֻמּוֹת הָעוֹלָם נִמְשָׁךְ מִתַּחַת הַזְּמַן, וְכֹחַ אֱלֹקִי הַמְּהַוֶּה הַטֶּבַע, מֻסְתָּר וְנֶעְלָם בְּשִׁבְעִים שָׂרִים, בְּסוֹד גָּלוּת הַשְּׁכִינָה כַּיָּדוּעַ.]{dir="rtl"}

[אֲבָל יִשְׂרָאֵל עָלָה בַּמַּחֲשָׁבָה, פֵּירוּשׁ, שֹׁרֶשׁ נִשְׁמָתָם וּמְקוֹר חֻצְבָם מִבְּחִינַת מַחֲשָׁבָה עֶלְיוֹנָה שֶׁלְּמַעְלָה מִן הַזְּמַן, וְעַל דֶּרֶךְ מָשָׁל הַמַּחֲשָׁבָה בָּאָדָם הַתַּחְתּוֹן שֶׁנּוֹבַעַת בִּתְמִידוּת בְּלִי הֶפְסֵק לְעוֹלָם, כָּךְ בְּחִינַת אוֹתִיּוֹת מֹחַ עֶלְיוֹנָה הָאֱלֹקִית כְּתִיב בָּהּ "וְנָהָר יֹצֵא מֵעֵדֶן", שֶׁהֵם בְּחִינַת חָכְמָה וּבִינָה, שֶׁנִּקְרָאִים תְּרֵין רֵעִין דְּלָא מִתְפָּרְשִׁין, פֵּירוּשׁ, שֶׁמְּחַדֵּשׁ הַשֶּׁפַע תָּמִיד מֵאַיִן לְיֵשׁ בִּבְלִי הֶפְסֵק כוּ', וְהוּא לְמַעְלָה מִן הַזְּמַן.]{dir="rtl"}

[יִשְׂרָאֵל עָלָה בַּמַּחֲשָׁבָה זוֹ, אֵין צָרִיךְ כְּלָל לִרְאָיָה וּמוֹפֵת לְחִידּוּשׁ הַהֲשְׁפָּעָה מֵאַיִן לְיֵשׁ, כְּמוֹ בְּקַיִץ וָחֹרֶף כַּנַּ"ל, כִּי מֻשֶּׂגֶת לָהֶם בְּנַפְשָׁם מִצַּד שֹׁרֶשׁ נְשָׁמוֹת בְּנֵי יִשְׂרָאֵל בִּבְחִינַת מַחֲשָׁבָה עֶלְיוֹנָה.]{dir="rtl"}

[וְזֶהוּ שֶׁאָנוּ אוֹמְרִים **מוֹדִים אֲנַחְנוּ לָךְ שֶׁאַתָּה הוּא ה' אֱלֹקֵינוּ**, פֵּירוּשׁ, מֵחֲמַת שֶׁאַתָּה הוּא ה' אֱלֹקֵינוּ בִּבְחִינַת מַחֲשָׁבָה עֶלְיוֹנָה, שֶׁעָלוּ שָׁם כְּנֶסֶת יִשְׂרָאֵל בְּשֹׁרֶשׁ נִשְׁמָתָם וּמְקוֹר חֻצְבָם, עַל כֵּן מִמֵּילָא אֲנַחְנוּ מוֹדִים בְּהוֹדָאָה זוֹ וּבְלִי שׁוּם מוֹפֵת וּרְאָיָה כְּלָל.]{dir="rtl"}

[**אַךְ** הִנֵּה, זֹאת הַהוֹדָאָה אֵינָהּ בְּנֶפֶשׁ הָאֱלֹקִית אֶלָּא מִצַּד שָׁרְשָׁהּ לְמַעְלָה בַּמַּחֲשָׁבָה, אֲבָל בִּהְיוֹתָהּ מְלֻבֶּשֶׁת בְּחֹמֶר הַגּוּפָנִי הַמַּחְשִׁיךְ עָלֶיהָ, לֹא תוּכַל לֵבָּא לִבְחִינַת הוֹדָאָה זֹאת שֶׁאֵין לְיֵשׁ מֵאַיִן לִהְיוֹת מְחֻדָּשׁ מֵאֱלֹקֵינוּ בִּתְמִידוּת כוּ', מִפְּנֵי שֶׁחֹמְרִיּוּת הַגּוּף מַחְשִׁיךְ עַל הַהוֹדָאָה הַזֹּאת.]{dir="rtl"}

[וְעִנְיַן מוֹדִים דְּרַבָּנָן, הוּא עִנְיַן הַהוֹדָאָה לְהוֹדָאָה זוֹ, כְּלוֹמַר, מוֹדִים אֲנַחְנוּ לָךְ עַל שֶׁאָנוּ מוֹדִים לָךְ כּוּ', אָנוּ מוֹדִים שֶׁכֵּן יָאָתָה לְהוֹדוֹת לָךְ כּוּ'.]{dir="rtl"}

**[וּבֵיאוּר הַדְּבָרִים, כִּי הִנֵּה יֵשׁ שְׁתֵּי מַדְרֵגוֹת בְּהוֹדָאָה:]{dir="rtl"}**

[**הָאַחַת** — הַהוֹדָאָה אֲשֶׁר בָּאָה מִצַּד הַהַשָּׂגָה וְהַדַּעַת, כְּמוֹ אָדָם שֶׁמַּשִּׂיג וּמַכִּיר בְּעֹצֶם גְּדוּלַּת וְרוֹמְמוּת הַמֶּלֶךְ וְכֵיצַד יֵדַע בַּהוֹדָאָה שֶׁכֵּן הוּא מֵצַד שֶׁמַּכִּיר וּמֵבִין כוּ', וְכֵן בְּחָכְמָה שֶׁיּוּדֶה בַּהַפְלָאַת חָכְמָתוֹ מֵצַד שֶׁיַּכִּיר וְיַשִּׂיג אוֹתוֹ.]{dir="rtl"}

[**הַשֵּׁנִית** — וְהִיא הַהוֹדָאָה שֶׁלְּמַטָּה מִן הַדַּעַת, וְהַיְינוּ כַּאֲשֶׁר אֵין בִּיכוֹלַת הַשָּׂגָתוֹ לְהַשִּׂיג וּלְהַכִּיר כְּלָל בְּעֹצֶם הַגְּדוּלָּה, רַק שֶׁיּוֹדֵעַ וּמַכִּיר שֶׁהוּא מוּפְלָא בְּחָכְמָה, וְצָרִיךְ לְהוֹדוֹת לוֹ בַּהוֹדָאָה וּלְשַׁבֵּחַ גָּדוֹל, וְכָמוֹ שֶׁיּוּדֶה גַּם עָנִי בְּגִדְּלוּת הַמֶּלֶךְ אַף עַל פִּי שֶׁאֵינוֹ מַכִּיר בִּמְהוּת הַגְּדוּלָּה כְּמוֹ שֶׁהִיא, אֶלָּא שֶׁמּוֹדֶה שֶׁצָּרִיךְ לְהוֹדוֹת, וּלְשַׁבֵּחַ בְּגִדְּלוּת זוֹ, הַגַּם שֶׁאֵינוֹ מַשִּׂיגָהּ, וְזֶה בָּא מִצַּד רִיחוּק מֵצַּד הַמּוֹדֶה, שֶׁכָּל שֶׁהוּא יוֹתֵר מְרוּחָק בְּעֶרֶךְ...]{dir="rtl"}

## נְקֻדּוֹת מַפְתֵּחַ

**[תְּשׁוּבָה לְאֻמּוֹת הָעוֹלָם]{dir="rtl"}**

> [זֶה הַמּוֹפֵת וְהָרְאָיָה מִקַּיִץ וְחוֹרֶף אֵינוֹ אֶלָּא לְאֻמּוֹת הָעוֹלָם, שֶׁנִּקְרָאִים עַם נָבָל]{dir="rtl"}

**[שׇׁרְשָׁם שֶׁל הַגּוֹיִם מִתַּחַת הַזְּמַן]{dir="rtl"}**

> [שֹׁרֶשׁ נַפְשׁוֹת אֻמּוֹת הָעוֹלָם נִמְשָׁךְ מִתַּחַת הַזְּמַן, וְכֹחַ אֱלֹקִי הַמְּהַוֶּה הַטֶּבַע מֻסְתָּר וְנֶעְלָם בְּשִׁבְעִים שָׂרִים, בְּסוֹד גָּלוּת הַשְּׁכִינָה]{dir="rtl"}

**[יִשְׂרָאֵל עָלָה בַּמַּחֲשָׁבָה]{dir="rtl"}**

> [שֹׁרֶשׁ נִשְׁמָתָם וּמְקוֹר חֻצְבָם מִבְּחִינַת מַחֲשָׁבָה עֶלְיוֹנָה שֶׁלְּמַעְלָה מִן הַזְּמַן]{dir="rtl"}

**[אֵין צָרִיךְ רְאָיָה וּמוֹפֵת]{dir="rtl"}**

> [יִשְׂרָאֵל עָלָה בַּמַּחֲשָׁבָה זוֹ, אֵין צָרִיךְ כְּלָל לִרְאָיָה וּמוֹפֵת לְחִידּוּשׁ הַהֲשְׁפָּעָה מֵאַיִן לְיֵשׁ]{dir="rtl"}

**[מוֹדִים אֲנַחְנוּ לָךְ]{dir="rtl"}**

> [מֵחֲמַת שֶׁאַתָּה הוּא ה' אֱלֹקֵינוּ בִּבְחִינַת מַחֲשָׁבָה עֶלְיוֹנָה... עַל כֵּן אֲנַחְנוּ מוֹדִים בְּהוֹדָאָה זוֹ בְּלִי שׁוּם מוֹפֵת וּרְאָיָה כְּלָל]{dir="rtl"}

**[הַחֹמְרִיּוּת מַחְשִׁיךְ עַל הַהוֹדָאָה]{dir="rtl"}**

> [בִּהְיוֹתָהּ מְלֻבֶּשֶׁת בְּחֹמֶר הַגּוּפָנִי הַמַּחְשִׁיךְ עָלֶיהָ, לֹא תוּכַל לֵבָּא לִבְחִינַת הוֹדָאָה זֹאת]{dir="rtl"}
