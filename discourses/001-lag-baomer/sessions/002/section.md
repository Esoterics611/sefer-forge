## הַנֹּסַח הָעִקָּרִי

...[מִפְּנֵי שֶׁחֹמְרִיּוּת הַגּוּף מַחְשִׁיךְ עַל הַהוֹדָאָה הַזֹּאת.]{dir="rtl"}

<!-- UNCLEAR: ~2 lines — transition sentence -->

[וְעִנְיַן מוֹדִים דְּרַבָּנָן, הוּא עִנְיַן הַהוֹדָאָה לְהוֹדָאָה זוֹ, כְּלוֹמַר, מוֹדִים אֲנַחְנוּ לָךְ עַל שֶׁאָנוּ מוֹדִים לָךְ כּוּ', אָנוּ מוֹדִים שֶׁכֵּן יָאָתָה לְהוֹדוֹת לָךְ כּוּ'.]{dir="rtl"}

**[וּבֵיאוּר הַדְּבָרִים, כִּי הִנֵּה יֵשׁ בּ' מַדְרֵגוֹת בְּהוֹדָאָה]{dir="rtl"}**²⁵. [הָאַחַת — הַהוֹדָאָה אֲשֶׁר בָּאָה מִצַּד הַהַשָּׂגָה וְהַדַּעַת, כְּמוֹ אָדָם שֶׁמַּשִּׂיג וּמַכִּיר בְּעֹצֶם גְּדוּלַּת וְרוֹמְמוּת הַמֶּלֶךְ וְכֵיצַד יֵדַע בַּהוֹדָאָה שֶׁכֵּן הוּא מֵצַד שֶׁמַּכִּיר וּמֵבִין כּוּ', וְכֵן בְּחָכָם שֶׁיּוֹדֶה בְּהַפְלָאַת חָכְמָתוֹ מִצַּד שֶׁיַּכִּיר וְיַשִּׂיג אוֹתוֹ²⁶.]{dir="rtl"}

[וְאָמְנָם יֵשׁ בְּחִינָה לְמַטָּה מִזֶּה, וְהוּא הַהוֹדָאָה שֶׁלְּמַטָּה מִן הַדַּעַת, דְּהַיְינוּ כַּאֲשֶׁר אֵין בִּיכוֹלַת הַשָּׂגָתוֹ לְהַשִּׂיג וּלְהַכִּיר כְּלָל בְּעֹצֶם הַגְּדוּלָּה, רַק שֶׁיּוֹדֵעַ וּמַכִּיר שֶׁהוּא מוּפְלָא בְּחָכְמָה, וְצָרִיךְ לְהוֹדוֹת לוֹ בַּהוֹדָאָה וּלְשַׁבֵּחַ גָּדוֹל, וּכְמוֹ שֶׁיּוֹדֶה גַּם הֶעָנִי בְּגִדְּלוּת הַמֶּלֶךְ אַף עַ"פ שֶׁאֵינוֹ מַכִּיר בְּמַהוּת הַגְּדוּלָּה כְּמוֹ שֶׁהִיא, אֶלָּא שֶׁמּוֹדֶה שֶׁצָּרִיךְ לְהוֹדוֹת, וּלְשַׁבֵּחַ בִּגְדוּלָּה זוֹ, הַגַּם שֶׁאֵינוֹ מַשִּׂיגָהּ, וְזֶה בָּא מִצַּד רִיחוּק מֵצַּד הַמּוֹדֶה²⁷, שֶׁכָּל שֶׁהוּא יוֹתֵר מְרוּחָק בְּעֵרֶךְ...]{dir="rtl"}

## נְקֻדּוֹת מַפְתֵּחַ

**[שְׁתֵּי מַדְרֵגוֹת בְּהוֹדָאָה]{dir="rtl"}**

> [וּבֵיאוּר הַדְּבָרִים, כִּי הִנֵּה יֵשׁ בּ' מַדְרֵגוֹת בְּהוֹדָאָה: הָאַחַת — מִצַּד הַהַשָּׂגָה וְהַדַּעַת; הַשֵּׁנִית — הוֹדָאָה שֶׁלְּמַטָּה מִן הַדַּעַת]{dir="rtl"}

**[נָעוּץ תְּחִלָּתָן בְּסוֹפָן]{dir="rtl"}**

> [נָעוּץ תְּחִלָּתָן בְּסוֹפָן וְסוֹפָן בִּתְחִלָּתָן — כָּל הָאוֹר הַיּוֹתֵר עֶלְיוֹן לֹא יוּכַל לָבֹא בְּגִלּוּי לְמַטָּה כִּי אִם בַּמַּדְרֵגָה הַיּוֹתֵר תַּחְתּוֹנָה]{dir="rtl"}

**[עֵינָא סַגִּיאָה]{dir="rtl"}**

> [הַהוֹדָאָה הַיּוֹתֵר תַּחְתּוֹנָה בִּקְדוּשָׁה — שֶׁבְּהוֹד שֶׁבְּהוֹד — בָּא דַוְקָא הַגִּלּוּי הַיּוֹתֵר עֶלְיוֹן]{dir="rtl"}

**[כַּלְלוּת עִנְיַן לַ"ג בָּעֹמֶר]{dir="rtl"}**

> [כַּלְלוּת סְפִירַת הָעוֹמֶר — לְבָרֵר וְלִזְקֹק אֶת הַמִּדּוֹת הָרְעוֹת דְּנֶפֶשׁ הַבְּהֵמִית, שֶׁיִּנָּקְתָהּ מֵאוֹמֶר שְׂעוֹרִים מַאֲכַל בְּהֵמָה]{dir="rtl"}
